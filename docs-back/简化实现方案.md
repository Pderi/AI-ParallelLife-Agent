# 三层记忆+RAG联动Agent 秋招极简实现方案（2周可落地）


 “我设计的三层记忆 + RAG 方案核心是‘分层管理 + 单向闭环’：热记忆用 Redis 存核心配置，保证每次对话必加载且极速响应；温记忆用 PostgreSQL 做结构化知识存储，结合向量检索精准召回用户专属知识；冷记忆存全量原始数据，作为知识提炼的唯一源头。整个流程遵循‘冷→温→热’单向流动，所有提炼操作异步执行，既保证用户体验，又能沉淀高价值的个性化知识，同时通过容量限制、多租户隔离等规则，确保系统能落地到生产环境。”  

**核心原则**：砍掉所有生产级冗余，只保留能体现架构能力和工程思维的核心逻辑，用最少的代码跑通完整闭环，面试时能讲清设计思路和解决的痛点。

## 一、项目整体定位
面向垂直领域（推荐做**个人开发者助手**，最容易演示和出效果）的智能Agent，通过「热记忆常驻+温记忆按需召回+冷记忆全量归档+RAG专业补充」，解决传统聊天机器人"长对话断片、无长期记忆、千人一面"的核心痛点，实现"越用越懂你"的个性化体验。

## 二、极简技术栈（一套数据库搞定所有）
| 层级 | 技术选型 | 选型理由 |

## 三、核心架构设计（简化版）
```plain
用户提问 → 对话服务 → 热记忆加载 → 温记忆召回 → RAG召回 → 上下文融合 → LLM生成回答
                                                          ↓
                                                    冷记忆落盘（原始对话）
                                                          ↓
                                                异步触发LLM提炼温记忆
                                                          ↓
                                                    更新热记忆全局索引
```

**核心铁则（面试必背）**：

1. 数据单向流动：冷→温→热，禁止反向写入
2. 优先级：热记忆 > 温记忆 > RAG
3. 所有原始数据先写冷记忆，再更新上层
4. 所有耗时操作异步执行，不阻塞对话

## 四、数据库设计（4张核心表，直接复制执行）
```sql
-- 1. 用户表（多租户隔离基础）
CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(50) NOT NULL UNIQUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    last_active_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. 热记忆表（核心配置+全局索引）
CREATE TABLE hot_memory (
    user_id UUID PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    global_rules TEXT NOT NULL DEFAULT '', -- 用户全局核心规则
    session_context TEXT NOT NULL DEFAULT '', -- 当前会话上下文（最近3轮）
    global_index JSONB NOT NULL DEFAULT '{}' -- 温记忆全局索引 {topic_name: scene}
);

-- 3. 温记忆表（用户专属结构化知识）
CREATE TABLE warm_memory (
    topic_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    topic_name VARCHAR(100) NOT NULL, -- 场景名称，如"Java代码规范"
    content TEXT NOT NULL, -- Markdown格式的知识内容
    scene VARCHAR(100) NOT NULL, -- 生效场景，如"Java后端开发"
    importance INT NOT NULL DEFAULT 3, -- 重要性1-5分
    content_vector vector(1536) NOT NULL, -- 向量，用于语义检索
    is_archived BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(user_id, topic_name)
);
-- 向量检索索引（必加）
CREATE INDEX idx_warm_vector ON warm_memory USING hnsw (content_vector vector_cosine_ops);

-- 4. 冷记忆表（全量原始对话）
CREATE TABLE cold_memory (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    session_id UUID NOT NULL,
    query TEXT NOT NULL,
    answer TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
-- 会话查询索引
CREATE INDEX idx_cold_session ON cold_memory(user_id, session_id);

-- 5. RAG知识库表（通用专业知识）
CREATE TABLE rag_knowledge (
    doc_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    doc_title VARCHAR(200) NOT NULL,
    doc_chunk TEXT NOT NULL, -- 单块200-500字
    doc_vector vector(1536) NOT NULL,
    is_valid BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
-- RAG向量索引
CREATE INDEX idx_rag_vector ON rag_knowledge USING hnsw (doc_vector vector_cosine_ops);
```

### 2. 核心工具函数
```javascript
const { pgPool, llm, embeddings } = require('./config');

// 生成文本向量
async function getVector(text) {
  const res = await embeddings.embeddings.create({
    model: 'text-embedding-3-small',
    input: text
  });
  return res.data[0].embedding;
}

// 初始化新用户记忆
async function initUserMemory(userId) {
  await pgPool.query(
    `INSERT INTO hot_memory (user_id) VALUES ($1) ON CONFLICT DO NOTHING`,
    [userId]
  );
}
```

### 3. 检索流程（用户提问→生成回答）
```javascript
// 步骤1：加载热记忆
async function getHotMemory(userId, sessionId) {
  // 加载全局规则和索引
  const hotRes = await pgPool.query(
    `SELECT global_rules, global_index FROM hot_memory WHERE user_id = $1`,
    [userId]
  );
  
  // 加载当前会话最近3轮对话
  const sessionRes = await pgPool.query(
    `SELECT query, answer FROM cold_memory 
     WHERE user_id = $1 AND session_id = $2 
     ORDER BY created_at DESC LIMIT 3`,
    [userId, sessionId]
  );
  
  const sessionContext = sessionRes.rows.reverse().map(row =>
    `用户：${row.query}\nAI：${row.answer}`
  ).join('\n');
  
  return {
    globalRules: hotRes.rows[0]?.global_rules || '',
    globalIndex: hotRes.rows[0]?.global_index || {},
    sessionContext
  };
}

// 步骤2：召回温记忆（Top3）
async function recallWarmMemory(userId, query) {
  const queryVector = await getVector(query);
  const res = await pgPool.query(
    `SELECT topic_name, content FROM warm_memory
     WHERE user_id = $1 AND is_archived = false
     ORDER BY content_vector <-> $2 LIMIT 3`,
    [userId, `[${queryVector.join(',')}]`]
  );
  
  return res.rows.map(row => `> ${row.topic_name}：${row.content}`).join('\n');
}

// 步骤3：召回RAG（Top3）
async function recallRAG(query) {
  const queryVector = await getVector(query);
  const res = await pgPool.query(
    `SELECT doc_title, doc_chunk FROM rag_knowledge
     WHERE is_valid = true
     ORDER BY doc_vector <-> $1 LIMIT 3`,
    [`[${queryVector.join(',')}]`]
  );
  
  return res.rows.map(row => `> ${row.doc_title}：${row.doc_chunk}`).join('\n');
}

// 步骤4：生成最终回答
async function generateAnswer(userId, sessionId, query) {
  await initUserMemory(userId);
  
  // 并行加载所有内容
  const [hotMemory, warmContent, ragContent] = await Promise.all([
    getHotMemory(userId, sessionId),
    recallWarmMemory(userId, query),
    recallRAG(query)
  ]);
  
  // 拼接Prompt（固定顺序，不能乱）
  const prompt = `
【系统设定】你是一个专业的个人开发者助手，帮助用户解决编程问题。
【热记忆·全局核心规则】
${hotMemory.globalRules}
【温记忆·用户专属知识】
${warmContent}
【RAG·专业知识库】
${ragContent}
【当前会话上下文】
${hotMemory.sessionContext}
【用户当前提问】
${query}
  `;
  
  // 调用LLM生成回答
  const llmRes = await llm.chat.completions.create({
    model: 'deepseek-chat',
    temperature: 0.7,
    messages: [{ role: 'user', content: prompt }]
  });
  const answer = llmRes.choices[0].message.content;
  
  // 写入冷记忆，触发后续提炼
  await appendColdMemory(userId, sessionId, query, answer);
  
  return answer;
}
```

### 4. 写入提炼流程（回答后→记忆沉淀）
```javascript
// 步骤1：写入冷记忆（必须先执行）
async function appendColdMemory(userId, sessionId, query, answer) {
  await pgPool.query(
    `INSERT INTO cold_memory (user_id, session_id, query, answer)
     VALUES ($1, $2, $3, $4)`,
    [userId, sessionId, query, answer]
  );
  
  // 异步触发温记忆提炼（延迟1分钟，给用户反馈时间）
  setTimeout(() => {
    extractColdToWarm(userId, sessionId);
  }, 60000);
  
  // 更新用户活跃时间
  await pgPool.query(
    `UPDATE users SET last_active_at = NOW() WHERE user_id = $1`,
    [userId]
  );
}

// 步骤2：冷→温记忆提炼（核心闭环）
async function extractColdToWarm(userId, sessionId) {
  // 读取本次会话所有对话
  const coldRes = await pgPool.query(
    `SELECT query, answer FROM cold_memory 
     WHERE user_id = $1 AND session_id = $2`,
    [userId, sessionId]
  );
  
  if (coldRes.rows.length < 2) return; // 对话太少不提炼
  
  const rawText = coldRes.rows.map(row =>
    `用户：${row.query}\nAI：${row.answer}`
  ).join('\n');
  
  // 调用LLM提炼结构化知识
  const extractPrompt = `
从以下对话中提取用户的个性化规则、偏好、踩坑记录，只保留可长期复用的内容。
要求：
1. 只提取用户明确说的内容，不要添加通用知识
2. 输出JSON格式，没有有效内容就输出{"has_memory": false}
3. 每条规则不超过50字

【输出格式】
{
  "has_memory": true,
  "topic_name": "场景名称",
  "scene": "生效场景",
  "content": "提炼的知识内容（Markdown列表）",
  "importance": 1-5
}

【对话内容】
${rawText}
  `;
  
  try {
    const extractRes = await llm.chat.completions.create({
      model: 'deepseek-chat',
      temperature: 0.1,
      response_format: { type: 'json_object' },
      messages: [{ role: 'user', content: extractPrompt }]
    });
    
    const result = JSON.parse(extractRes.choices[0].message.content);
    if (!result.has_memory) return;
    
    // 生成向量
    const vector = await getVector(result.content);
    
    // 写入或更新温记忆
    await pgPool.query(
      `INSERT INTO warm_memory (user_id, topic_name, content, scene, importance, content_vector)
       VALUES ($1, $2, $3, $4, $5, $6)
       ON CONFLICT (user_id, topic_name) DO UPDATE
       SET content = $3, scene = $4, importance = $5, content_vector = $6, updated_at = NOW()`,
      [userId, result.topic_name, result.content, result.scene, result.importance, `[${vector.join(',')}]`]
    );
    
    // 更新热记忆全局索引
    await pgPool.query(
      `UPDATE hot_memory 
       SET global_index = jsonb_set(global_index, '{${result.topic_name}}', $1)
       WHERE user_id = $2`,
      [result.scene, userId]
    );
    
    console.log(`温记忆提炼成功：${result.topic_name}`);
  } catch (err) {
    console.error('温记忆提炼失败：', err);
  }
}
```

# 三层记忆+平行宇宙人生模拟器 结合价值（秋招专属版）
**核心结论**：三层记忆架构能把"一次性的人生模拟工具"升级为**"懂你的专属人生规划师"**，彻底解决普通AI模拟器"千人一面、记不住用户、无法持续迭代"的致命痛点，同时让你的项目技术深度直接拉满，成为秋招中碾压普通CRUD和简单RAG项目的杀器。

## 一、用户体验价值：从"通用模拟"到"专属人生顾问"
### 1. 彻底解决"每次都要重新自我介绍"的痛点
普通人生模拟器：用户每次开新对话都要重新输入"我25岁、程序员、工作2年、月薪15k"，体验极差，无法持续使用。  
**三层记忆解决**：

+ **热记忆**：永久存储用户核心画像（年龄、职业、学历、城市、薪资、核心诉求），100%每次对话自动加载，用户再也不用重复说明自己的情况
+ **温记忆**：记住用户的历史选择、价值观偏好（比如"更看重工作生活平衡而非高薪"、"不接受996"、"对创业有执念"），模拟时自动适配用户的性格和价值观
+ **冷记忆**：存储所有历史模拟报告和对话记录，用户可以随时回溯"3个月前我模拟的创业路径是什么样的"

**示例**：  
用户第一次："我25岁，程序员，工作2年，想转行产品经理"  
系统生成报告后，温记忆自动提炼：`用户偏好：工作生活平衡>高薪；转行动机：厌倦写代码；风险承受能力：中等`  
用户第二次："帮我看看回老家发展的可能性"  
系统自动加载用户画像，直接生成"25岁上海程序员回老家西安发展"的专属报告，无需任何重复输入

### 2. 实现"越用越懂你"的持续迭代体验
普通人生模拟器：每次模拟都是独立的，不会从用户的反馈中学习，永远是通用模板。  
**三层记忆解决**：

+ **冷→温提炼闭环**：自动从用户的对话和反馈中提炼个性化规则，比如用户说"我不喜欢太激进的建议"，温记忆会自动记录，后续所有模拟都会偏向保守稳健的路径
+ **温记忆反哺模拟**：随着用户使用次数增加，系统积累的用户专属知识越来越多，模拟结果会越来越精准，真正成为"最懂你的人生规划师"
+ **跨会话记忆继承**：用户无论开多少个对话窗口，所有的偏好、历史、画像都会全局继承，实现"一次输入，终身使用"

### 3. 支持"渐进式人生探索"
普通人生模拟器：只能做单次、孤立的决策模拟，无法模拟"连续决策"的连锁反应。  
**三层记忆解决**：

+ 热记忆记录当前探索的"人生时间线"，用户可以在一个宇宙中继续做后续决策
+ 比如：先模拟"转行产品经理"→再模拟"转行后3年跳槽"→再模拟"跳槽后创业"，系统会记住每一步的结果，生成连贯的人生路径
+ 冷记忆存储所有探索过的宇宙，用户可以随时切换不同的平行宇宙，对比不同选择的长期影响

# AI自动判断调用冷记忆的完整实现方案（秋招简化版）
**核心设计思路**：在「热→温→RAG」标准检索流程后，增加**AI信息充足性自检环节**，让LLM先判断现有上下文是否足够回答问题；只有当明确信息不足时，才自动触发冷记忆检索→结构化提炼→上下文补充的完整流程。全程对用户透明，无需手动触发。

## 一、核心逻辑与触发条件
### 1. 为什么不能每次都调用冷记忆？
+ 冷记忆存储全量原始对话，数据量大，检索和提炼耗时久（秒级），会显著增加用户等待时间
+ 原始数据包含大量无关信息，直接提炼会引入噪音，污染上下文
+ 会产生额外的LLM Token消耗，增加成本

### 2. 自动触发冷记忆的核心场景（结合人生模拟器）
AI会在以下情况自动判断需要调用冷记忆：

| 场景类型 | 具体示例 |
| --- | --- |
| 历史信息回溯 | "3个月前我模拟的创业路径里，第5年的关键事件是什么？" |
| 跨会话关联 | "上次我问过回老家西安的可能性，这次帮我对比一下成都的情况" |
| 细节补充需求 | "你之前说我转行产品经理的成功率是65%，这个数字是怎么算出来的？" |
| 上下文冲突 | "你刚才说我月薪15k，但我之前明明说过我现在月薪20k" |
| 深度分析需求 | "把我所有探索过的路径做一个综合对比，给我最终建议" |


### 3. 完整流程设计
```plain
用户提问 → 加载热记忆 → 召回温记忆 → 召回RAG → 【AI信息充足性判断】
                                                          ↓
                        ┌─────────────────────────────┐
                        │ 现有信息足够？               │
                        └─────┬─────────────────┬─────┘
                              │是               │否
                              ↓                 ↓
                        生成最终回答       检索相关冷记忆（按用户+关键词+时间）
                                                  ↓
                                            LLM提炼核心信息
                                                  ↓
                                            补充到上下文
                                                  ↓
                                            生成最终回答
                                                  ↓
                                            写入冷记忆，触发温记忆提炼
```

## 二、代码实现（完全兼容你之前的三层记忆代码）
### 1. 第一步：实现AI信息充足性判断函数
这是整个功能的核心，通过结构化Prompt让LLM输出明确的判断结果。

```javascript
const { llm } = require('./config');

/**
 * AI判断是否需要调用冷记忆
 * @param {string} query 用户当前提问
 * @param {object} context 现有上下文（热+温+RAG）
 * @returns {boolean} 是否需要调用冷记忆
 */
async function judgeNeedColdMemory(query, context) {
  const judgePrompt = `
你是一个信息充足性判断专家。请判断基于以下现有上下文，是否足够回答用户的问题。

【判断规则】
1. 只有当现有上下文**完全无法**回答问题，或者**缺少关键历史信息**时，才返回true
2. 如果现有信息足够，或者问题只需要基于通用知识回答，返回false
3. 必须严格输出JSON格式，不能有其他内容

【需要调用冷记忆的情况】
- 用户询问之前对话中提到过的具体历史信息
- 用户要求对比之前的模拟结果或历史选择
- 用户提到了"上次"、"之前"、"3个月前"等历史时间词
- 现有上下文存在矛盾，需要回溯原始对话确认
- 用户要求基于所有历史对话进行综合分析

【现有上下文】
热记忆（用户核心画像）：${context.globalRules}
温记忆（用户专属知识）：${context.warmContent}
RAG（通用知识）：${context.ragContent}
当前会话上下文：${context.sessionContext}

【用户当前提问】
${query}

【输出格式】
{"need_cold": true/false, "reason": "简短说明原因"}
  `;

  try {
    const res = await llm.chat.completions.create({
      model: 'deepseek-chat',
      temperature: 0.1,
      response_format: { type: 'json_object' },
      messages: [{ role: 'user', content: judgePrompt }]
    });
    
    const result = JSON.parse(res.choices[0].message.content);
    console.log(`冷记忆调用判断：${result.need_cold}，原因：${result.reason}`);
    return result.need_cold;
  } catch (err) {
    console.error('冷记忆判断失败：', err);
    return false; // 判断失败时默认不调用，保证系统稳定性
  }
}
```

### 2. 第二步：实现冷记忆检索与提炼函数
冷记忆不能直接进入上下文，必须先检索相关片段，再由LLM提炼出核心信息。

```javascript
const { pgPool, llm, embeddings } = require('./config');

/**
 * 检索并提炼冷记忆中的相关信息
 * @param {string} userId 用户ID
 * @param {string} query 用户当前提问
 * @returns {string} 提炼后的冷记忆核心信息
 */
async function retrieveAndRefineColdMemory(userId, query) {
  // 步骤1：检索相关冷记忆（按关键词+时间范围过滤，避免全量扫描）
  // 秋招简化版：先按用户ID检索最近3个月的对话，再用向量相似度排序
  const queryVector = await embeddings.embedQuery(query);
  const coldRes = await pgPool.query(
    `SELECT query, answer, created_at FROM cold_memory
     WHERE user_id = $1 
     AND created_at >= NOW() - INTERVAL '3 months'
     ORDER BY (to_tsvector('english', query || ' ' || answer) <=> to_tsvector('english', $2)) DESC
     LIMIT 10`, // 最多取10条相关对话
    [userId, query]
  );

  if (coldRes.rows.length === 0) {
    return '【冷记忆检索结果】未找到相关历史信息';
  }

  // 步骤2：拼接原始冷记忆片段
  const rawColdContent = coldRes.rows.map(row =>
    `[${row.created_at.toLocaleString()}]
用户：${row.query}
AI：${row.answer}
---`
  ).join('\n');

  // 步骤3：LLM提炼核心信息（只保留和当前问题相关的内容）
  const refinePrompt = `
从以下历史对话中，提炼出和用户当前提问最相关的核心信息。
要求：
1. 只保留和问题直接相关的内容，删除所有无关信息
2. 语言简洁，不超过300字
3. 如果没有相关内容，输出"无相关历史信息"

【用户当前提问】
${query}

【历史对话片段】
${rawColdContent}

【提炼结果】
  `;

  try {
    const refineRes = await llm.chat.completions.create({
      model: 'deepseek-chat',
      temperature: 0.1,
      messages: [{ role: 'user', content: refinePrompt }]
    });
    
    const refinedContent = refineRes.choices[0].message.content.trim();
    return `【冷记忆补充信息】\n${refinedContent}`;
  } catch (err) {
    console.error('冷记忆提炼失败：', err);
    return '【冷记忆检索结果】信息提炼失败';
  }
}
```

## 三、秋招简化优化与工程保障
### 1. 性能优化（避免用户等待过久）

```javascript
// 给冷记忆调用加超时
const coldPromise = retrieveAndRefineColdMemory(userId, query);
const timeoutPromise = new Promise((resolve) => {
  setTimeout(() => resolve('【冷记忆检索超时】'), 3000);
});
coldContent = await Promise.race([coldPromise, timeoutPromise]);
```

+ **缓存机制**：同一个会话中，相同问题的冷记忆结果缓存5分钟，避免重复调用
+ **检索范围限制**：默认只检索最近3个月的冷记忆，用户明确要求更早内容时再扩大范围

### 2. 错误处理与熔断
+ 连续3次冷记忆调用失败后，自动熔断10分钟，期间不再调用冷记忆

+ 所有冷记忆相关操作都加try-catch，避免单个环节失败导致整个对话崩溃

3. **信息提炼机制**：冷记忆原始数据不直接进入上下文，必须经过LLM提炼，避免了噪音污染和Token爆炸
